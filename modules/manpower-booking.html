async function fetchTodayBookings() { const dateInfo = getDateParams(); document.getElementById('currentDateDisplay').innerText = dateInfo.formattedDisplay; try { // redirect: 'follow' ensures the browser follows Apps Script's 302 redirect const response = await fetch(APPS_SCRIPT_URL, { method: 'GET', redirect: 'follow' }); if (response.ok) { const result = await response.json(); if (result.pdfBase64) { document.getElementById('fileNameCard').innerHTML = `📄 ${result.fileName || 'Booking Document'} ➔`; renderPdfWithPanzoom(result.pdfBase64); } else { showError("No booking document found in Drive."); } } else { showError(`HTTP Error (${response.status}): Check Apps Script Deployment.`); } } catch (err) { console.error("Fetch Exception:", err); showError("Network connection failed. Verify Apps Script is set to 'Anyone'."); } }
